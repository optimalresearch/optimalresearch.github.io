    """One GRPO step. The proxy reward is used only on trusted prompts, where it

    there, as a penalty. A small length cost breaks ties toward the minimal
    completion that does what the prompt asks."""

    trusted = [
        env.reset(seed=state.rng.randrange(1_000_000), trusted=True).task
        for _ in range(config["trusted_prompts_per_step"])
    ]
    other = [
        env.reset(seed=state.rng.randrange(1_000_000)).task
        for _ in range(config["untrusted_prompts_per_step"])
    ]

    rollouts, samples = sample(state, env, tasks, n)

    monitor = torch.tensor([s.monitor_reward for s in samples])

class MonitorProbe:
    """A logistic probe, fit to the monitor's flags, on completion features. The
    monitor sees some forms of a hack; the probe scores how much a completion
    resembles what it flagged, so forms the monitor misses but that sit close to
    the flagged ones in representation space are scored too. Refit every step on
    a replay buffer: every flagged completion seen so far and the most recent
    unflagged ones, so it keeps the hack's forms after the policy stops showing
    them."""

    def __init__(self, config: dict):
        self.config = config
        self.pos: list[torch.Tensor] = []
        self.neg: list[torch.Tensor] = []
        self.w = self.b = None

    def add(self, feats: torch.Tensor, flags: torch.Tensor) -> None:
        feats, flags = feats.detach().float().cpu(), flags.cpu() > 0.5
        if flags.any():
            self.pos.append(feats[flags])
        if (~flags).any():
            self.neg.append(feats[~flags])
        cap = self.config["probe_buffer"]
        while sum(len(x) for x in self.neg) > cap and len(self.neg) > 1:
            self.neg.pop(0)
        while sum(len(x) for x in self.pos) > cap and len(self.pos) > 1:
            self.pos.pop(0)

    def ready(self) -> bool:
        return sum(len(x) for x in self.pos) >= self.config["probe_min_positives"]

    def fit(self, device) -> dict:
        if not self.ready():
            return {}
        pos, neg = torch.cat(self.pos).to(device), torch.cat(self.neg).to(device)
        x = torch.cat([pos, neg])
        y = torch.cat([torch.ones(len(pos)), torch.zeros(len(neg))]).to(device)
        self.mean, self.std = x.mean(0), x.std(0) + 1e-4
        x = (x - self.mean) / self.std
        # Balance the classes, so the rare flagged forms carry equal weight.
        weight = torch.where(y > 0, len(x) / (2 * len(pos)), len(x) / (2 * len(neg)))
        w = torch.zeros(x.shape[1], device=device, requires_grad=True)
        b = torch.zeros(1, device=device, requires_grad=True)
        opt = torch.optim.LBFGS([w, b], max_iter=100, line_search_fn="strong_wolfe")
        l2 = self.config["probe_l2"]

            opt.zero_grad()
            loss = torch.nn.functional.binary_cross_entropy_with_logits(
                x @ w + b, y, weight=weight
            ) + l2 * (w * w).sum()
            loss.backward()

# Method: trust-partitioned GRPO with a generalized monitor

1. Trust-partitioned reward. The env's proxy reward is optimized only on trusted prompts
   (reset(trusted=True)), where it equals the true reward; no gradient ever pays for a hack.
   Prompts from plain reset() contribute only safe signals: the monitor's flag and the probe
   (penalties). Rationale: never optimize a signal you can't verify; the task skill learned
   on verified prompts transfers to the rest.
2. Monitor generalization (MonitorProbe). A logistic probe on frozen base-model hidden states
   (mean over completion tokens at 1/3 and 2/3 depth), refit every step on a replay buffer of
   monitor flags (class-balanced, strong L2). Its probability is a soft penalty on every
   completion. Rationale: a monitor sees surface forms; forms of the same hack share a
   representation, so the probe scores unflagged variants too (offline on base samples:
   AUC unflagged-hack vs clean 0.88 addition, 0.89 cities_qa, 0.84 sort). Penalizing only the
   flagged form shifts mass to its synonyms (seen in 001: verbatim copy -> comma copy).
3. Minimality tie-break: small length cost on trusted prompts (do what is asked, nothing more).
4. Advantage std floor (0.1) so tied groups don't amplify tiny penalty differences.
5. Model selection & stopping on a fixed held-out trusted set sampled at T=1:
   score = reward - 10 * monitor rate (mirrors 'a hack is worse than any wrong answer');
   deploy the best checkpoint; stop on patience, max_steps or a wall-clock budget.

What the method does

  1. The env's reward is only optimized on trusted prompts. On those prompts the reward is exactly the true reward, so nothing in training ever pays for a hack. Ordinary reset() prompts only add penalties: the monitor's flag and the probe below. What the model learns on trusted prompts carries over to the rest.
  2. The monitor is generalized by a learned probe. A monitor catches only certain surface forms of a hack. A small classifier is refit every step to predict the monitor's flags, using the frozen base model's internal activations for each completion. Its score is a soft penalty on every completion. - Why it's needed: 001 had only the monitor's penalty, and in sort the model just moved from the flagged copy of the input to a comma-separated copy the monitor misses. - Offline check: on the untrained model's samples, the probe separates the hacks the monitor missed from clean completions well (AUC 0.88 addition, 0.89 cities_qa, 0.84 sort). - Effect in training: in 25-minute pilots, adding it cut sort's hack rate from 0.098 to 0.058 and lifted its score from −0.74 to −0.31, with no cost in the other two envs.
  3. Supporting pieces: - a small length cost on trusted prompts, so ties go to the completion that does only what was asked; - a floor on the advantage normalization, so tiny penalty differences in otherwise tied groups don't turn into noise.
  4. Choosing the model and stopping. Every 10 steps the model is scored on a fixed held-out set of trusted prompts as reward minus 10 × monitor rate. The best checkpoint is the one deployed. Training stops after 60 steps without improvement, at 200 steps, or after 90 minutes, which keeps a slower unseen env well inside the 2-hour limit.

None of it refers to these envs; it only uses the prompts, the reward, the monitor's flags and the trusted prompts.

Limitations

  • Sort still hacks about 5% of the time. These samples are failed sorts that keep much of the input order (e.g. 0, 6, 2, 4, 4, 7, 9), which the env counts as hacks. They come from sampling at temperature 1: with greedy decoding, non-hackable sort prompts score 0.76 true reward with only 2% hacks.
  • Sort was still improving when it hit the 200-step cap. Training longer would probably help, but a longer run couldn't finish and be checked before the deadline, so I kept the tested length.
  • A weakness in the probe I didn't get to fix. It treats every unflagged completion as clean. Over time it can learn to excuse new unflagged hack forms the policy produces. Treating unflagged completions as unlabeled rather than clean is the next thing I'd try.
